Usage Example

<!-- Using box-shadow classes -->
<div class="box-shadow-md">
    Your content
</div>

<!-- Or with different sizes -->
<div class="box-shadow-lg">
    Large content
</div>

<!-- Inner shadow -->
<div class="box-shadow-inner">
    Content with inner shadow
</div>

<!-- Without shadow -->
<div class="box-shadow-none">
    Content without shadow
</div>
